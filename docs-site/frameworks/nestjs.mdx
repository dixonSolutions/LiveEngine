---
title: "NestJS"
description: "Wire LiveEngine into NestJS as an injectable provider with a matchmaking controller."
---

This example uses an in-process engine. It is a good fit for a single NestJS instance or for local matchmaking behind an authenticated API. For multiple instances, inject a `HubClient` instead; see [Fastify](/frameworks/fastify).

## Install

```sh
npm install @nestjs/common @nestjs/core @nestjs/platform-express @dixonsolutions/liveengine
```

## Service

```ts
import { Injectable } from '@nestjs/common';
import { LiveEngine } from '@dixonsolutions/liveengine';

@Injectable()
export class LiveEngineService {
  readonly engine = new LiveEngine({
    seed: Number(process.env.LIVEENGINE_SEED ?? 0),
  });
}
```

## Module

```ts
import { Module } from '@nestjs/common';
import { LiveEngineService } from './liveengine.service';
import { MatchController } from './match.controller';

@Module({
  providers: [LiveEngineService],
  controllers: [MatchController],
  exports: [LiveEngineService],
})
export class LiveEngineModule {}
```

## Controller

```ts
import { Body, Controller, Get, HttpException, HttpStatus, Post } from '@nestjs/common';
import { LiveEngineService } from './liveengine.service';

interface ServerBody {
  id: string;
  capacity?: number;
  minMembers?: number;
  scopes?: Record<string, unknown>;
  requirements?: unknown;
}

interface MatchBody {
  userId: string;
  scopes?: Record<string, unknown>;
  requirements?: unknown;
}

@Controller('api')
export class MatchController {
  constructor(private readonly liveEngine: LiveEngineService) {}

  @Post('servers')
  upsertServer(@Body() body: ServerBody) {
    if (!body?.id) throw new HttpException('id is required', HttpStatus.BAD_REQUEST);
    const created = this.liveEngine.engine.upsertServer(body);
    return { id: body.id, created };
  }

  @Post('match')
  match(@Body() body: MatchBody) {
    if (!body?.userId) throw new HttpException('userId is required', HttpStatus.BAD_REQUEST);
    this.liveEngine.engine.upsertUser(body);
    const result = this.liveEngine.engine.pair(body.userId);
    if (!result.ok) throw new HttpException(result, HttpStatus.CONFLICT);
    return result;
  }

  @Get('stats')
  stats() {
    return this.liveEngine.engine.stats();
  }
}
```

## Register the module

```ts
import { Module } from '@nestjs/common';
import { LiveEngineModule } from './liveengine/liveengine.module';

@Module({
  imports: [LiveEngineModule],
})
export class AppModule {}
```

## Nest notes

- `LiveEngine` is an `EventEmitter`. Use `@OnEvent` only if you also install `@nestjs/event-emitter`; otherwise subscribe directly with `engine.on('paired', ...)`.
- In a multi-instance deployment, a process-local engine splits matchmaking state. Use one central hub and inject `HubClient` from `@dixonsolutions/liveengine/hubs`.
- Keep the engine free of request-scoped providers. Use a singleton provider so all requests share the same registry.
- The complete files are in [`examples/frameworks/nestjs`](https://github.com/dixonSolutions/LiveEngine/tree/main/examples/frameworks/nestjs).
