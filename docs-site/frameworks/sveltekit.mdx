SvelteKit server routes keep the hub token in the server runtime. The page below polls the room list and connects as a client.

## Shared server client

`src/lib/server/liveengine.js`

```js
import { env } from '$env/dynamic/private';
import { HubClient } from '@dixonsolutions/liveengine/hubs';

let client;

export function liveEngine() {
  if (!env.LIVEENGINE_URL || !env.LIVEENGINE_TOKEN) {
    throw new Error('Set LIVEENGINE_URL and LIVEENGINE_TOKEN');
  }
  client ??= new HubClient(env.LIVEENGINE_URL, {
    token: env.LIVEENGINE_TOKEN,
    timeoutMs: 10_000,
  });
  return { client, hubId: env.LIVEENGINE_HUB_ID || 'ranked' };
}
```

## Server listing

`src/routes/api/liveengine/servers/+server.js`

```js
import { json } from '@sveltejs/kit';
import { liveEngine } from '$lib/server/liveengine';

export async function GET() {
  const { client, hubId } = liveEngine();
  const [servers, stats] = await Promise.all([
    client.call(hubId, 'listServers'),
    client.call(hubId, 'stats'),
  ]);
  return json({ hubId, servers, stats });
}
```

## Matchmaking

`src/routes/api/liveengine/match/+server.js`

```js
import { json } from '@sveltejs/kit';
import { liveEngine } from '$lib/server/liveengine';

export async function POST({ request }) {
  const body = await request.json().catch(() => null);
  if (!body?.userId) return json({ error: 'userId is required' }, { status: 400 });

  const { client, hubId } = liveEngine();
  await client.call(hubId, 'upsertUser', [{
    id: body.userId,
    scopes: body.scopes ?? {},
    requirements: body.requirements,
  }]);
  const result = await client.call(hubId, 'autoJoin', [body.userId]);
  if (!result.ok) return json(result, { status: 409 });
  const server = await client.call(hubId, 'getServer', [result.serverId]);
  return json({ ...result, server });
}
```

## Native ping

`src/routes/api/liveengine/ping/+server.js`

```js
import { json } from '@sveltejs/kit';
import { liveEngine } from '$lib/server/liveengine';

export async function GET() {
  const { client } = liveEngine();
  return json(await client.ping({ samples: 5 }));
}
```

## Client page

`src/routes/liveengine/+page.svelte`

```svelte
<script>
  import { onMount } from 'svelte';

  let userId = 'player-1';
  let servers = [];
  let stats = null;
  let ping = null;
  let message = '';

  async function load() {
    const response = await fetch('/api/liveengine/servers', { cache: 'no-store' });
    const data = await response.json();
    if (!response.ok) return (message = data.error || 'Could not load servers');
    servers = data.servers ?? [];
    stats = data.stats;
  }

  async function connect() {
    message = 'Connecting...';
    const response = await fetch('/api/liveengine/match', {
      method: 'POST',
      headers: { 'content-type': 'application/json' },
      body: JSON.stringify({ userId, scopes: { region: 'au' }, requirements: {} }),
    });
    const data = await response.json();
    message = response.ok
      ? `Matched ${userId} into ${data.serverId}`
      : data.error || data.reason || 'No match';
    load();
  }

  async function samplePing() {
    const response = await fetch('/api/liveengine/ping', { cache: 'no-store' });
    const data = await response.json();
    if (response.ok) ping = data;
    else message = data.error || 'Ping failed';
  }

  onMount(() => {
    load();
    const timer = setInterval(load, 2500);
    return () => clearInterval(timer);
  });
</script>

<main>
  <h1>LiveEngine matchmaking</h1>
  <div class="actions">
    <input bind:value={userId} aria-label="User ID">
    <button on:click={connect}>Find a match</button>
    <button on:click={samplePing}>Ping hub</button>
  </div>
  <p>{message}</p>
  {#if ping}
    <p>Ping {ping.avgMs.toFixed(1)} ms average ({ping.jitterMs.toFixed(1)} ms jitter)</p>
  {/if}
  {#if stats}
    <p>{stats.paired} paired / {stats.waiting} waiting / {stats.open} open rooms</p>
  {/if}
  <h2>Servers</h2>
  <ul>
    {#each servers as server}
      <li>
        <strong>{server.id}</strong> — {server.status}, {server.members.length}/{server.capacity} joined,
        {server.free} free — <code>{JSON.stringify(server.scopes)}</code>
      </li>
    {/each}
  </ul>
</main>

<style>
  main { max-width: 880px; margin: 40px auto; padding: 20px; font-family: system-ui, sans-serif; }
  .actions { display: flex; gap: 8px; margin-bottom: 16px; }
</style>
```
